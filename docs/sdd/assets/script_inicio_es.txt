
CREATE TABLE AccionAgente
(
  idAccion             BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  idConversacion       BIGINT                   NOT NULL,
  idMensajeOrigen      BIGINT                  ,
  tipoAccion           VARCHAR(100)             NOT NULL,
  parametros           JSONB                    NOT NULL DEFAULT '{}'::jsonb,
  estado               VARCHAR(30)              NOT NULL,
  fechaSolicitud       TIMESTAMP WITH TIME ZONE NOT NULL,
  fechaEjecucion       TIMESTAMP WITH TIME ZONE,
  resultado            JSONB                   ,
  requiereConfirmacion BOOLEAN                  NOT NULL DEFAULT false,
  fechaConfirmacion    TIMESTAMP WITH TIME ZONE,
  PRIMARY KEY (idAccion)
);

COMMENT ON TABLE AccionAgente IS 'Acción interpretada o ejecutada por el agente, con trazabilidad y confirmación.';

COMMENT ON COLUMN AccionAgente.idMensajeOrigen IS 'FK opcional al mensaje que originó la acción.';

COMMENT ON COLUMN AccionAgente.parametros IS 'Parámetros estructurados validados antes de ejecutar.';

COMMENT ON COLUMN AccionAgente.estado IS 'Enum JPA sugerido: PROPUESTA, PENDIENTE_CONFIRMACION, EJECUTADA, FALLIDA, CANCELADA.';

COMMENT ON COLUMN AccionAgente.resultado IS 'Resultado estructurado o detalle del error de ejecución.';

COMMENT ON COLUMN AccionAgente.fechaConfirmacion IS 'Momento en que el usuario confirmó una acción sensible.';

CREATE TABLE Asignacion
(
  idAsignacion          BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  tituloAsignacion      VARCHAR(150)             NOT NULL,
  descripcionAsignacion TEXT                    ,
  fechaAsignacion       TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
  fechaLimiteAsignacion TIMESTAMP WITH TIME ZONE,
  estadoAsignacion      VARCHAR(30)              NOT NULL DEFAULT 'ACTIVA',
  idUsuarioCreador      BIGINT                   NOT NULL,
  idEquipo              BIGINT                   NOT NULL,
  PRIMARY KEY (idAsignacion)
);

COMMENT ON TABLE Asignacion IS 'Asignación de práctica. Los destinatarios se registran explícitamente para conservar el alcance histórico.';

COMMENT ON COLUMN Asignacion.tituloAsignacion IS 'Nombre visible de la práctica o conjunto de problemas.';

COMMENT ON COLUMN Asignacion.estadoAsignacion IS 'Enum: ACTIVA, FINALIZADA, CANCELADA.';

COMMENT ON COLUMN Asignacion.idUsuarioCreador IS 'Usuario que creó la asignación; debe ser coach autorizado del equipo.';

CREATE TABLE AsignacionDestinatario
(
  idAsignacion      BIGINT NOT NULL,
  idMembresiaEquipo BIGINT NOT NULL,
  PRIMARY KEY (idAsignacion, idMembresiaEquipo)
);

COMMENT ON TABLE AsignacionDestinatario IS 'Destinatarios explícitos de una asignación; preserva el alcance aunque cambie la membresía del equipo.';

CREATE TABLE Competencia
(
  idCompetencia         BIGINT       NOT NULL GENERATED ALWAYS AS IDENTITY,
  nombreCompetencia     VARCHAR(150) NOT NULL,
  fechaCompetencia      DATE         NOT NULL,
  plataformaCompetencia VARCHAR(80)  NOT NULL,
  PRIMARY KEY (idCompetencia)
);

CREATE TABLE ConversacionIA
(
  idConversacion         BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  idUsuario              BIGINT                   NOT NULL,
  fechaInicio            TIMESTAMP WITH TIME ZONE NOT NULL,
  fechaUltimaInteraccion TIMESTAMP WITH TIME ZONE NOT NULL,
  estado                 VARCHAR(30)              NOT NULL,
  titulo                 VARCHAR(200)            ,
  PRIMARY KEY (idConversacion)
);

COMMENT ON TABLE ConversacionIA IS 'Sesión de conversación entre un usuario y el asistente de IA.';

COMMENT ON COLUMN ConversacionIA.idConversacion IS 'PK autogenerada; compatible con GenerationType.IDENTITY.';

COMMENT ON COLUMN ConversacionIA.idUsuario IS 'FK obligatoria al usuario propietario de la conversación.';

COMMENT ON COLUMN ConversacionIA.estado IS 'Enum JPA sugerido: ACTIVA, CERRADA, ARCHIVADA.';

CREATE TABLE CuentaExterna
(
  idCuentaExterna         BIGINT       NOT NULL GENERATED ALWAYS AS IDENTITY,
  plataformaCuentaExterna VARCHAR(50)  NOT NULL,

CREATE TABLE DetalleAsignacion
(

CREATE TABLE DisponibilidadUsuario
(

  horaInicio              TIME        NOT NULL,
  horaFin                 TIME        NOT NULL,
  PRIMARY KEY (idDisponibilidadUsuario)
);

COMMENT ON TABLE DisponibilidadUsuario IS 'Disponibilidad estructurada del usuario para comparar con horarios de equipos.';

COMMENT ON COLUMN DisponibilidadUsuario.horaFin IS 'Debe ser posterior a horaInicio.';

CREATE TABLE Equipo
(
  idEquipo          BIGINT       NOT NULL GENERATED ALWAYS AS IDENTITY,
  nombreEquipo      VARCHAR(120) NOT NULL,
  nivelEquipo       VARCHAR(30)  NOT NULL,
  descripcionEquipo TEXT        ,
  estadoEquipo      VARCHAR(30)  NOT NULL DEFAULT 'ACTIVO',
  idUsuarioCoach    BIGINT       NOT NULL,
  PRIMARY KEY (idEquipo)
);

COMMENT ON COLUMN Equipo.estadoEquipo IS 'Enum de aplicación: ACTIVO, INACTIVO, CERRADO.';

COMMENT ON COLUMN Equipo.idUsuarioCoach IS 'Usuario responsable del equipo; el backend debe validar rol COACH activo.';

CREATE TABLE HorarioEquipo
(
  idHorarioEquipo BIGINT      NOT NULL GENERATED ALWAYS AS IDENTITY,

  diaSemana       VARCHAR(15) NOT NULL,
  horaInicio      TIME        NOT NULL,
  horaFin         TIME        NOT NULL,
  PRIMARY KEY (idHorarioEquipo)
);

COMMENT ON TABLE HorarioEquipo IS 'Horario estructurado del equipo para filtros y búsqueda por lenguaje natural.';

COMMENT ON COLUMN HorarioEquipo.horaFin IS 'Debe ser posterior a horaInicio.';

CREATE TABLE MembresiaEquipo
(
  idMembresiaEquipo           BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  fechaIngresoMembresiaEquipo TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
  estadoMembresiaEquipo       VARCHAR(30)              NOT NULL DEFAULT 'ACTIVA',
  idUsuario                   BIGINT                   NOT NULL,
  idEquipo                    BIGINT                   NOT NULL,
  fechaSalidaMembresiaEquipo  TIMESTAMP WITH TIME ZONE,
  PRIMARY KEY (idMembresiaEquipo)
);

COMMENT ON TABLE MembresiaEquipo IS 'Período histórico de pertenencia. Debe existir como máximo una membresía ACTIVA por usuario y equipo mediante índice único parcial.';

COMMENT ON COLUMN MembresiaEquipo.estadoMembresiaEquipo IS 'Enum: ACTIVA, RETIRADA, EXPULSADA.';

COMMENT ON COLUMN MembresiaEquipo.idUsuario IS 'Integrante con rol PRACTICANTE.';

CREATE TABLE MensajeIA
(
  idMensaje      BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  idConversacion BIGINT                   NOT NULL,
  rol            VARCHAR(20)              NOT NULL,
  contenido      TEXT                     NOT NULL,
  fecha          TIMESTAMP WITH TIME ZONE NOT NULL,
  metadatos      JSONB                   ,
  PRIMARY KEY (idMensaje)
);

COMMENT ON TABLE MensajeIA IS 'Mensaje persistido del usuario, asistente, sistema o herramienta.';

COMMENT ON COLUMN MensajeIA.rol IS 'Enum JPA sugerido: USER, ASSISTANT, SYSTEM, TOOL.';

COMMENT ON COLUMN MensajeIA.metadatos IS 'Datos opcionales del proveedor, tokens, citas o invocaciones de herramientas.';

CREATE TABLE PerfilCompetitivo

CREATE TABLE Problema

  tituloProblema      VARCHAR(200) NOT NULL,
  descripcionProblema TEXT        ,

);

CREATE TABLE RecursoAcademico
(
  idRecursoAcademico     BIGINT       NOT NULL GENERATED ALWAYS AS IDENTITY,
  tituloRecursoAcademico VARCHAR(200) NOT NULL,
  autorRecursoAcademico  VARCHAR(150),

  PRIMARY KEY (idRecursoAcademico)
);

CREATE TABLE RecursoAcademicoTema
(

  PRIMARY KEY (idRecursoAcademico, idTema)
);

COMMENT ON TABLE RecursoAcademicoTema IS 'Clasificación temática del recurso para sugerencias cuando no existe un vínculo directo con un problema.';

CREATE TABLE Resolucion
(
  idResolucion                  BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  fechaResolucion               TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
  estadoResolucion              VARCHAR(30)              NOT NULL DEFAULT 'PENDIENTE',
  puntajeResolucion             DOUBLE PRECISION        ,
  urlEvidenciaResolucion        VARCHAR(500)            ,
  idMembresiaEquipo             BIGINT                   NOT NULL,
  idDetalleAsignacion           BIGINT                   NOT NULL,
  plataformaEvidenciaResolucion VARCHAR(50)             ,
  identificadorEnvioResolucion  VARCHAR(150)            ,
  fechaValidacionResolucion     TIMESTAMP WITH TIME ZONE,
  PRIMARY KEY (idResolucion)
);

COMMENT ON COLUMN Resolucion.estadoResolucion IS 'Enum: PENDIENTE, VALIDADA, RECHAZADA.';

COMMENT ON COLUMN Resolucion.urlEvidenciaResolucion IS 'URL del envío en la plataforma externa.';

COMMENT ON COLUMN Resolucion.idDetalleAsignacion IS 'Detalle asignado que fue resuelto.';

CREATE TABLE ResultadoCompetencia
(
  idResultadoCompetencia                 BIGINT           NOT NULL GENERATED ALWAYS AS IDENTITY,
  puestoResultadoCompetencia             INTEGER         ,
  puntajeResultadoCompetencia            DOUBLE PRECISION,
  problemasResueltosResultadoCompetencia INTEGER          NOT NULL DEFAULT 0,
  estadoResultadoCompetencia             VARCHAR(30)      NOT NULL DEFAULT 'PENDIENTE',
  idEquipo                               BIGINT           NOT NULL,
  idCompetencia                          BIGINT           NOT NULL,
  PRIMARY KEY (idResultadoCompetencia)
);

CREATE TABLE SolicitudIngreso
(
  idSolicitudIngreso             BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  fechaSolicitudIngreso          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
  estadoSolicitudIngreso         VARCHAR(30)              NOT NULL DEFAULT 'PENDIENTE',
  observacionSolicitudIngreso    TEXT                    ,
  idUsuario                      BIGINT                   NOT NULL,
  idEquipo                       BIGINT                   NOT NULL,
  fechaRespuestaSolicitudIngreso TIMESTAMP WITH TIME ZONE,
  idUsuarioRevisor               BIGINT                  ,
  PRIMARY KEY (idSolicitudIngreso)
);

COMMENT ON TABLE SolicitudIngreso IS 'Solicitud de ingreso. Debe existir como máximo una solicitud PENDIENTE por usuario y equipo mediante índice único parcial en la migración PostgreSQL.';

COMMENT ON COLUMN SolicitudIngreso.estadoSolicitudIngreso IS 'Enum: PENDIENTE, ACEPTADA, RECHAZADA, CANCELADA.';

COMMENT ON COLUMN SolicitudIngreso.idUsuario IS 'Solicitante con rol PRACTICANTE.';

COMMENT ON COLUMN SolicitudIngreso.idUsuarioRevisor IS 'Usuario COACH que procesó la solicitud.';

CREATE TABLE Tema
(
  idTema          BIGINT       NOT NULL GENERATED ALWAYS AS IDENTITY,
  nombreTema      VARCHAR(100) NOT NULL UNIQUE,
  descripcionTema TEXT        ,
  PRIMARY KEY (idTema)
);

CREATE TABLE Usuario
(
  idUsuario            BIGINT                   NOT NULL GENERATED ALWAYS AS IDENTITY,
  nombreUsuario        VARCHAR(80)              NOT NULL,
  apellidoUsuario      VARCHAR(80)              NOT NULL,
  correoUsuario        VARCHAR(255)             NOT NULL UNIQUE,
  passwordHashUsuario  VARCHAR(255)             NOT NULL,
  estadoUsuario        VARCHAR(30)              NOT NULL DEFAULT 'ACTIVO',
  rolUsuario           VARCHAR(30)              NOT NULL,
  fechaRegistroUsuario TIMESTAMP WITH TIME ZONE NOT NULL,
  PRIMARY KEY (idUsuario)
);

COMMENT ON COLUMN Usuario.correoUsuario IS 'Correo único utilizado para autenticación.';

COMMENT ON COLUMN Usuario.passwordHashUsuario IS 'Hash de contraseña; nunca almacenar contraseña en texto plano.';

COMMENT ON COLUMN Usuario.estadoUsuario IS 'Enum de aplicación: ACTIVO, INACTIVO, BLOQUEADO.';

COMMENT ON COLUMN Usuario.rolUsuario IS 'Rol único del usuario: PRACTICANTE, COACH o ADMIN. El rol COACH requiere verificación antes de activarse.';
